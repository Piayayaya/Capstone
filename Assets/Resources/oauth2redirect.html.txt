  <script>
    // Example final app scheme (must match your manifest + DeepLinkReceiver)
    var schemeUrl = "brainyme:/oauth2redirect";

    // Take everything after "#" from Google (id_token, state, etc.)
    var hash = window.location.hash;  // e.g. #id_token=...&state=...

    if (hash && hash.length > 1) {
      // Redirect phone to the app with the SAME fragment
      window.location = schemeUrl + hash;
    } else {
      document.body.innerHTML = "<p>No token in URL. Please try again.</p>";
    }
  </script>
